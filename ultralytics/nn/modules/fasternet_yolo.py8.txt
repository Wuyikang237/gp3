import torch
import torch.nn as nn
from ultralytics.nn.modules import Detect, C2f, Conv
from ultralytics.utils.torch_utils import initialize_weights
from fasternet import FasterNet
from ultralytics.utils.loss import v8DetectionLoss
class FasterNetYOLO(nn.Module):
    def __init__(self, nc=2, ch=3, embed_dim=96, depths=(1,2,8,2), mlp_ratio=2., **kwargs):
        super().__init__()
        self.nc = nc
        self.ch = ch
        self.end2end = False
        self.stride = torch.tensor([32.])   # 占位，会在第一次前向时更新
        self.args = {}
        self.names = {i: f"{i}" for i in range(nc)}
        # 构建 FasterNet 主干
        self.backbone = FasterNet(
            in_chans=ch,
            embed_dim=embed_dim,
            depths=depths,
            mlp_ratio=mlp_ratio,
            fork_feat=True,
            **kwargs
        )
        
        # 计算各阶段通道数
        c2 = embed_dim * 2   # 8倍下采样
        c3 = embed_dim * 4   # 16倍下采样
        c4 = embed_dim * 8   # 32倍下采样
        
        # 构建检测头
        self.head = self._build_head(c2, c3, c4, nc)

        # 损失函数（训练时使用）
        self.criterion = v8DetectionLoss(self)
        
        # 初始化权重（可以加载预训练）
        initialize_weights(self)
    
        # 训练器需要的属性
        self.end2end = False
        self.stride = torch.tensor([32.])   # 占位，后续会被检测头覆盖
        self.args = {}
        self.names = {i: f"{i}" for i in range(nc)}
        
       
    def _build_head(self, c2, c3, c4, nc):
        """构建 YOLOv8 风格的检测头，返回 ModuleDict"""
        conv_c2 = Conv(c2, 256, 1)
        conv_c3 = Conv(c3, 512, 1)
        conv_c4 = Conv(c4, 1024, 1)
        upsample = nn.Upsample(scale_factor=2, mode='nearest')
        c2f_1 = C2f(512, 512)
        c2f_2 = C2f(256, 256)
        c2f_3 = C2f(512, 512)
        c2f_4 = C2f(1024, 1024)
        down_1 = Conv(256, 256, 3, 2)
        down_2 = Conv(512, 512, 3, 2)
        detect = Detect(nc=nc, ch=(256, 512, 1024))
        return nn.ModuleDict({
            'conv_c2': conv_c2, 'conv_c3': conv_c3, 'conv_c4': conv_c4,
            'upsample': upsample, 'c2f_1': c2f_1, 'c2f_2': c2f_2,
            'c2f_3': c2f_3, 'c2f_4': c2f_4, 'down_1': down_1,
            'down_2': down_2, 'detect': detect,
        })
    
    def forward(self, x):
        # 获取主干特征
        features = self.backbone.forward_det(x)   # [f0, f1, f2, f3]
        f1, f2, f3 = features[1], features[2], features[3]  # P3, P4, P5
        
        p3 = self.head['conv_c2'](f1)   # 256
        p4 = self.head['conv_c3'](f2)   # 512
        p5 = self.head['conv_c4'](f3)   # 1024
        
        # 上采样融合 P5 和 P4
        up4 = self.head['upsample'](p5)
        cat4 = torch.cat([up4, p4], dim=1)
        x4 = self.head['c2f_1'](cat4)
        
        # 上采样融合 P4 和 P3
        up3 = self.head['upsample'](x4)
        cat3 = torch.cat([up3, p3], dim=1)
        x3 = self.head['c2f_2'](cat3)
        
        # 下采样融合 P3 到 P4
        down3 = self.head['down_1'](x3)
        cat_down4 = torch.cat([down3, x4], dim=1)
        x4_new = self.head['c2f_3'](cat_down4)
        
        # 下采样融合 P4 到 P5
        down4 = self.head['down_2'](x4_new)
        cat_down5 = torch.cat([down4, p5], dim=1)
        x5 = self.head['c2f_4'](cat_down5)
        
        # Detect 层
        out = self.head['detect']([x3, x4_new, x5])
        return out
